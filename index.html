Pengaturan Integrasi Google Sheets

Masukkan URL Web App yang dideploy dari Apps Script sekolah Anda.

Kode Google Apps Script (GAS) Reference:

Salin kode berikut ke Apps Script (Menu Ekstensi > Apps Script) di Google Spreadsheet Anda. Kode ini akan otomatis membuat header, format rupiah, dan tanda tangan:

// =========================================================================
// KODE GOOGLE APPS SCRIPT (GAS) - ARKAS BOS REGULER
// =========================================================================

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("ARKAS_BOS") || ss.getActiveSheet();
  var data = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "ARKAS_BOS";
    var sheet = ss.getSheetByName(sheetName);
    
    // 1. Otomatis Buat Sheet Baru Jika Belum Ada
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    }
    
    sheet.clear(); // Bersihkan data lama agar selalu terbarui presisi
    
    var payload = JSON.parse(e.postData.contents);
    var activities = payload.activities || payload;
    var meta = payload.meta || {};
    
    // 2. Tulis Header Identitas Sekolah & Ringkasan Pagu
    sheet.getRange("A1").setValue("RENCANA KEGIATAN DAN ANGGARAN SEKOLAH (ARKAS) BOS REGULER").setFontWeight("bold").setFontSize(14);
    sheet.getRange("A2").setValue("Nama Sekolah: " + (meta.schoolName || "SD NEGERI"));
    sheet.getRange("D2").setValue("NPSN / Kab: " + (meta.npsn || "-"));
    sheet.getRange("A3").setValue("Jumlah Siswa: " + (meta.siswa || 0) + " Siswa");
    sheet.getRange("D3").setValue("Tarif BOS/Siswa: Rp " + (meta.tarif || 0));
    sheet.getRange("G3").setValue("Total Pagu: Rp " + (meta.totalPagu || 0));

    // 3. Tulis Header Judul Tabel Utama
    var headers = ["NO", "URAIAN RENCANA KEGIATAN", "SATUAN", "HARGA SATUAN (RP)", "JAN (RP)", "FEB (RP)", "MAR (RP)", "APR (RP)", "MEI (RP)", "JUN (RP)", "JUL (RP)", "AGS (RP)", "SEP (RP)", "OKT (RP)", "NOV (RP)", "DES (RP)", "JUMLAH (RP)"];
    sheet.getRange(5, 1, 1, headers.length).setValues([headers]);
    
    // Styling Header Tabel (Latar Biru Kemdikbud & Teks Putih)
    var headerRange = sheet.getRange(5, 1, 1, headers.length);
    headerRange.setBackground("#005596")
               .setFontColor("#FFFFFF")
               .setFontWeight("bold")
               .setHorizontalAlignment("center")
               .setVerticalAlignment("middle");
    
    // 4. Masukkan Data Baris Kegiatan
    var startRow = 6;
    if (activities && activities.length > 0) {
      var rows = [];
      for (var i = 0; i < activities.length; i++) {
        var act = activities[i];
        if (Array.isArray(act)) {
          rows.push(act);
        } else {
          var bulanArr = act.bulan || new Array(12).fill(0);
          var harga = act.harga || 0;
          var row = [
            i + 1,
            act.nama,
            act.satuan || '-',
            harga
          ];
          var rowTotal = 0;
          for (var m = 0; m < 12; m++) {
            var cost = (bulanArr[m] || 0) * harga;
            row.push(cost);
            rowTotal += cost;
          }
          row.push(rowTotal);
          rows.push(row);
        }
      }
      
      sheet.getRange(startRow, 1, rows.length, headers.length).setValues(rows);
      
      // Format Otomatis Mata Uang Rupiah (Kolom D s/d Q)
      sheet.getRange(startRow, 4, rows.length, 14).setNumberFormat('"Rp "#,##0');
      
      // 5. Rumus Otomatis Total Alokasi Per Bulan (=SUM)
      var totalRowIndex = startRow + rows.length;
      sheet.getRange(totalRowIndex, 2).setValue("TOTAL ALOKASI PER BULAN").setFontWeight("bold");
      
      for (var col = 4; col <= 17; col++) {
        var colLetter = String.fromCharCode(64 + col);
        sheet.getRange(totalRowIndex, col).setFormula("=SUM(" + colLetter + startRow + ":" + colLetter + (totalRowIndex - 1) + ")")
             .setFontWeight("bold")
             .setNumberFormat('"Rp "#,##0');
      }
      
      sheet.getRange(totalRowIndex, 1, 1, headers.length).setBackground("#E6F0FA");
      
      // 6. Otomatis Membuat Lembar Tanda Tangan (Kepala Sekolah & Bendahara)
      var signRow = totalRowIndex + 3;
      sheet.getRange(signRow, 2).setValue("Mengetahui,");
      sheet.getRange(signRow, 14).setValue("Disetujui Oleh,");
      sheet.getRange(signRow + 1, 2).setValue("Kepala Sekolah");
      sheet.getRange(signRow + 1, 14).setValue("Bendahara BOS");
      
      sheet.getRange(signRow + 5, 2).setValue(meta.kepsekName || "-----------------").setFontWeight("bold");
      sheet.getRange(signRow + 6, 2).setValue("NIP. " + (meta.kepsekNip || "-"));
      
      sheet.getRange(signRow + 5, 14).setValue(meta.bendaharaName || "-----------------").setFontWeight("bold");
      sheet.getRange(signRow + 6, 14).setValue("NIP. " + (meta.bendaharaNip || "-"));
    }

    // Auto-fit ukuran kolom
    for (var k = 1; k <= headers.length; k++) {
      sheet.autoResizeColumn(k);
    }
    
    return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Data ARKAS berhasil disinkronkan & Sheet dibuat!"}))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}
